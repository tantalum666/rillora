import test from 'node:test';
import assert from 'node:assert/strict';
import worker, {probe} from './edge-probe.mjs';
const env={PROBE_SECRET:'synthetic-probe-secret-with-no-service-permissions'};
test('worker adapter does not pass its execution context as the fetch implementation', async()=>{
 const original=globalThis.fetch;
 try {
  globalThis.fetch=async()=>new Response('adapter-ok');
  const response=await worker.fetch(new Request('https://candidate.example/capabilities',
    {method:'POST',headers:{'x-probe-secret':env.PROBE_SECRET}}),env,{waitUntil(){}});
  assert.equal(response.status,200);assert.equal(await response.text(),'adapter-ok');
 } finally {globalThis.fetch=original;}
});
test('rejects missing configuration, unauthorized callers and non-canonical routes without forwarding',async()=>{
 let calls=0;const forward=()=>{calls++;throw Error('must not forward');};
 assert.equal((await probe(new Request('https://candidate.example/capabilities'),{},forward)).status,503);
 assert.equal((await probe(new Request('https://candidate.example/capabilities'),env,forward)).status,401);
 for(const [path,method] of [['/capabilities?target=https://evil.test','POST'],['/other','POST'],['/capabilities','GET']]) {
  assert.equal((await probe(new Request('https://candidate.example'+path,{method,headers:{'x-probe-secret':env.PROBE_SECRET}}),env,forward)).status,404);
 }
 assert.equal(calls,0);
});
test('uses only the fixed upstream and server credential, streams response and never follows redirects',async()=>{
 const request=new Request('https://candidate.example/capabilities',{method:'POST',headers:{'x-probe-secret':env.PROBE_SECRET,authorization:'must-not-forward'},body:'ignored'});
 const result=await probe(request,env,async(url,options)=>{
  assert.equal(new URL(url).hostname,'br-shiny-darkness-b9bgwknr-rilloraai.compute.c-7.eu-central-1.aws.neon.tech');
  assert.equal(options.redirect,'manual');assert.equal(options.body,'{}');assert.equal(options.headers.authorization,undefined);
  assert.equal(options.headers['x-probe-secret'],env.PROBE_SECRET);
  return new Response('protected-result',{status:200,headers:{'content-type':'application/json'}});
 });
 assert.equal(result.status,200);assert.equal(result.headers.get('cache-control'),'no-store');assert.equal(await result.text(),'protected-result');
});
