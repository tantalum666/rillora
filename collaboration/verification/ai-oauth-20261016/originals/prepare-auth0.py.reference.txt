NAME='Rillora MCP Acceptance - Codex';RESOURCE='http://127.0.0.1:58431/mcp';PURPOSE='mcp-synthetic-acceptance-20261016'
CLIENT={'name':NAME,'app_type':'native','oidc_conformant':True,'is_first_party':False,'token_endpoint_auth_method':'none','grant_types':['authorization_code'],'callbacks':[],
 'client_metadata':{'environment':'isolated-acceptance','purpose':PURPOSE},'jwt_configuration':{'alg':'RS256','lifetime_in_seconds':900}}

def save(path,value):
 path.write_text(json.dumps(value,indent=2)+'\n');path.chmod(0o600)
def signature(value):return hashlib.sha256(json.dumps(value,sort_keys=True,separators=(',',':')).encode()).hexdigest()
def existing_resources(values):
 return sorted([{k:v.get(k) for k in ['id','name','identifier','signing_alg','token_lifetime','token_lifetime_for_web','allow_offline_access','scopes']}for v in values if v['identifier']!=RESOURCE],key=lambda v:v['id'])

# Store only public registration metadata, never the unused secret included by provider create responses.

save(PRIVATE/'registration.json',state)
for connection in connections:
 if identifier in helpers.connection_clients(connection['id']):api('patch','connections/'+connection['id']+'/clients',[{'client_id':identifier,'status':False}])

current=api('get','clients/'+identifier+'?fields='+FIELDS+'&include_fields=true');helpers.match(current,CLIENT)

(QA/'auth0-preparation-result.json').write_text(json.dumps(report,indent=2)+'\n')
print(json.dumps({'applied':True,'verified':True,'ordinaryApplications':len([c for c in after_clients if not c.get('global',False)]),'existingClientsAndApisUnchanged':True,'existingBindingsUnchanged':True,'tenantSettingsUnchanged':True,'newClientInactive':True}))
