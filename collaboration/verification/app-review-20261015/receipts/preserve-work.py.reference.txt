        parent=Path(base)
        for name in list(dirs):
            path=parent/name;rel=path.relative_to(ROOT).as_posix();reason=directory_exclusion(rel)

            path=parent/name;rel=path.relative_to(ROOT).as_posix();reason=file_exclusion(rel)

        assert set(bundle.namelist())=={f.relative_to(source).as_posix()for f in source.rglob('*')if f.is_file()}
        assert all(bundle.read(name)==(source/name).read_bytes()for name in bundle.namelist())
    files.append((path.relative_to(ROOT).as_posix(),path))
