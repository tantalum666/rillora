import { readFileSync } from "node:fs";
import { createSecureContext } from "node:tls";
import pg from "pg";
import { contentClient, KmsContentKeyProvider } from "./private-content.js";
import { configureRecoveryPool, publishRecoveryTransaction, recoveryPublisherFromEnvironment } from './recovery-journal.js';
import { bindDatabaseAuthority, configureDatabaseAuthority } from './database-authority.js';
import { emailMatchKey } from './sharing-identity.js';
const { Pool } = pg;
const protection = new WeakMap();
export function configureContentPool(pool, config, preview = false, provider = config ? new KmsContentKeyProvider(config) : null) {
    protection.set(pool, { config, provider, preview });
}
export async function checkContentKeys(pool) {
    const settings = protection.get(pool);
    if (!settings?.config)
        return;
    if (!settings.provider)
        throw new Error("Private content key provider is unavailable");
    await settings.provider.check();
}
export function databaseConnectionOptions(config) {
    const connection = config.connectionString
        ? { connectionString: config.connectionString }
        : {
            host: config.host,
            port: config.port,
            database: config.database,
            user: config.user,
            password: config.password,
        };
    let ssl = false;
    if (config.sslMode === "verify-full") {
        const ca = readFileSync(config.sslCaPath, "utf8");
        // One context per pool avoids reparsing/retaining the large RDS root store
        // whenever an idle connection is replaced. Hostname/chain checks stay on.
        ssl = { rejectUnauthorized: true, ca, secureContext: createSecureContext({ ca }) };
    }
    return { ...connection, ssl };
}
export function createPool(config, testContentProvider) {
    if (testContentProvider && config.nodeEnv !== "test")
        throw new Error("Injected content keys are restricted to tests");
    const pool = new Pool({
        ...databaseConnectionOptions(config.database),
        max: config.databasePoolMax,
        idleTimeoutMillis: 30_000,
        connectionTimeoutMillis: config.databaseConnectionTimeoutMs,
        application_name: "continuity-api",
    });
    configureContentPool(pool, config.contentProtection ?? null, config.allowFamilyMaintenancePreview ?? false, testContentProvider ?? (config.contentProtection ? new KmsContentKeyProvider(config.contentProtection) : null));
    if (config.nodeEnv === 'production' && !config.databaseAuthorityKey)
        throw new Error('Production requires database transaction authority');
    if (config.nodeEnv === 'production')
        emailMatchKey(config);
    if (config.deletionWorkerEnabled && !config.deletionProof)
        throw new Error('Isolated deletion requires protected provider identity proofs');
    configureDatabaseAuthority(pool, config.databaseAuthorityKey ?? null);
    if (config.nodeEnv === 'production') {
        const journal = recoveryPublisherFromEnvironment();
        if (journal)
            configureRecoveryPool(pool, journal);
    }
    return pool;
}
async function protectedPlannerOperation(pool, client, operation) {
    const settings = protection.get(pool) ?? { config: null, provider: null, preview: false };
    const wrapped = contentClient(client, settings.config, settings.provider, settings.preview);
    try {
        const result = await operation(wrapped.client);
        await wrapped.session.prune();
        return result;
    }
    finally {
        wrapped.session.dispose();
    }
}
async function inRoleTransaction(pool, role, operation, repeatableSnapshot = false, plannerId = null) {
    const client = await pool.connect();
    try {
        await client.query(repeatableSnapshot ? "begin isolation level repeatable read" : "begin");
        await client.query(`set local role ${role}`);
        await bindDatabaseAuthority(pool, client, role, plannerId);
        const result = await operation(client);
        await publishRecoveryTransaction(pool, client);
        await client.query("commit");
        return result;
    }
    catch (error) {
        await client.query("rollback");
        throw error;
    }
    finally {
        client.release();
    }
}
export function withRecipientAcceptanceDatabase(pool, acceptanceTokenHash, newSessionTokenHash, operation) {
    return inRoleTransaction(pool, "continuity_rehearsal", async (client) => {
        await client.query("select set_config('app.rehearsal_acceptance_hash', $1, true), set_config('app.rehearsal_new_session_hash', $2, true), set_config('app.rehearsal_session_hash', $2, true)", [acceptanceTokenHash.toString("hex"), newSessionTokenHash.toString("hex")]);
        return operation(client);
    });
}
export function withRecipientRehearsalDatabase(pool, sessionTokenHash, operation) {
    return inRoleTransaction(pool, "continuity_rehearsal", async (client) => {
        await client.query("select set_config('app.rehearsal_session_hash', $1, true)", [
            sessionTokenHash.toString("hex"),
        ]);
        const contextResult = await client.query(`select id::text, public_id, planner_id::text, person_id::text, invitation_id::text
       from app.recipient_rehearsals
       where session_token_hash = $1 and status = 'accepted' and expires_at > now()`, [sessionTokenHash]);
        const row = contextResult.rows[0];
        if (!row)
            return null;
        await client.query("select set_config('app.current_rehearsal_id', $1, true), set_config('app.current_planner_id', $2, true)", [row.id, row.planner_id]);
        await bindDatabaseAuthority(pool, client, 'continuity_rehearsal', row.planner_id);
        return operation(client, {
            id: row.id,
            publicId: row.public_id,
            plannerId: row.planner_id,
            personId: row.person_id,
            invitationId: row.invitation_id,
        });
    });
}

export async function withDeletionWorkerDatabase(pool, operation) {
    const client = await pool.connect();
    try {
        await client.query('begin');
        await client.query('set local role continuity_deletion_worker');
        const result = await operation(client);
        await publishRecoveryTransaction(pool, client);
        await client.query('commit');
        return result;
    }
    catch (error) {
        await client.query('rollback');
        throw error;
    }
    finally {
        client.release();
    }
}

export async function withContentMigrationDatabase(pool, plannerId, operation) {
    return inRoleTransaction(pool, "continuity_app", async (raw) => {
        const allowed = (await raw.query("select pg_has_role(session_user,'continuity_content_service','member') as allowed")).rows[0]?.allowed;
        if (!allowed)
            throw new Error("Content migration requires the dedicated administrative credential");
        await raw.query('set local role continuity_content_service');
        await raw.query("select set_config('app.current_planner_id',$1,true),set_config('app.content_migration','true',true)", [plannerId]);
        await raw.query("select pg_advisory_xact_lock(hashtextextended('sharing:'||app.current_planner_id()::text,0))");
        const settings = protection.get(pool);
        if (!settings?.config || settings.config.mode !== "encrypted" || !settings.provider)
            throw new Error("Content migration requires the encrypted writer and key provider");
        const wrapped = contentClient(raw, settings.config, settings.provider, settings.preview);
        try {
            return await operation(raw, wrapped.client, wrapped.session);
        }
        finally {
            wrapped.session.dispose();
        }
    });
}

       from app.resolve_commerce_account($1::uuid)`, [commerceAccountReference]);

            return null;
        await client.query("select set_config('app.current_planner_id', $1, true)", [row.planner_id]);
        await bindDatabaseAuthority(pool, client, 'continuity_commerce', row.planner_id);

            commerceAccountId: row.commerce_account_id,
            plannerId: row.planner_id,
            planId: row.plan_id,
            publicId: row.public_id,
        });
    });
}
