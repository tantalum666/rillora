// Actual installed Codex OAuth/discovery acceptance; no provider tokens are exported.
import {spawn} from 'node:child_process';import {createInterface} from 'node:readline';
import {readFile,writeFile,appendFile,mkdtemp} from 'node:fs/promises';import {resolve} from 'node:path';import {homedir,tmpdir} from 'node:os';
const qa=resolve('output/qa/rillora-customer-mcp-20261018'),privateDir=resolve('.local-secrets/mcp-customer-20261018');
const name='rillora_customer_20261018',resource='https://app.rillora.xyz/api/mcp';
const source=await readFile(resolve(process.env.CODEX_HOME??resolve(homedir(),'.codex'),'config.toml'),'utf8');
const sections=[...source.matchAll(/^\[mcp_servers\.("(?:[^"\\]|\\.)+"|[A-Za-z0-9_-]+)\]\s*$/gm)].map(m=>m[1]);
const overrides=['-c','features.apps=false','-c','features.plugins=false','-c','features.remote_plugin=false','-c','features.multi_agent=false',
 '-c','features.multi_agent_v2=false','-c','features.shell_tool=false','-c','features.in_app_browser=false','-c','web_search="disabled"',
 ...sections.flatMap(s=>['-c',`mcp_servers.${s}.enabled=false`]),
 '-c',`mcp_servers.${name}.url="${resource}"`,'-c',`mcp_servers.${name}.enabled=true`,
 '-c',`mcp_servers.${name}.oauth.client_id="rillora-codex-v1"`,
 '-c',`mcp_servers.${name}.oauth.callback_url="http://127.0.0.1/callback"`,
 '-c',`mcp_servers.${name}.enabled_tools=["list_selected_matters","get_preparation_gaps"]`,
 '-c','mcp_oauth_credentials_store="keyring"'];
await writeFile(resolve(privateDir,'codex-test-overrides.json'),JSON.stringify(overrides),{mode:0o600});
const work=await mkdtemp(resolve(tmpdir(),'rillora-mcp-consumer-'));
await writeFile(resolve(privateDir,'codex-test-workdir.txt'),work,{mode:0o600});
let child,seq=0;const pending=new Map();
const rpc=(method,params)=>new Promise((resolve,reject)=>{const id=++seq;const timer=setTimeout(()=>{pending.delete(id);reject(Error('Client request timeout: '+method))},60000);pending.set(id,j=>{clearTimeout(timer);j.error?reject(Error('Client request rejected: '+method)):resolve(j.result)});child.stdin.write(JSON.stringify({jsonrpc:'2.0',id,method,params})+'\n')});
async function start(){
 child=spawn('codex',['app-server','--stdio',...overrides],{cwd:work,stdio:['pipe','pipe','pipe']});
 child.stderr.on('data',d=>void appendFile(resolve(privateDir,'codex-stderr.log'),d,{mode:0o600}));
 createInterface({input:child.stdout}).on('line',line=>{let j;try{j=JSON.parse(line)}catch{return;}
  if(pending.has(j.id)){pending.get(j.id)(j);pending.delete(j.id);}
  else if(j.method==='mcpServer/oauthLogin/completed'){
   void writeFile(resolve(qa,'codex-oauth-completion.json'),JSON.stringify({success:j.params.success,actualClient:true},null,2)+'\n');
   console.log(JSON.stringify({oauthCompleted:j.params.success}));
  }
 });
 await rpc('initialize',{clientInfo:{name:'rillora_customer_acceptance',version:'0.1.0'},capabilities:{experimentalApi:true,explicitGatewayOauth:true}});
 child.stdin.write(JSON.stringify({jsonrpc:'2.0',method:'initialized',params:{}})+'\n');
}
await start();console.log(JSON.stringify({ready:true,installedClient:'Codex',globalConfigUnchanged:true,otherExplicitServersDisabled:sections.length,commands:['login','status','restart-status','stop']}));
for await(const cmd of createInterface({input:process.stdin}))try{
 if(cmd==='login'){
  const result=await rpc('mcpServer/oauth/login',{name,scopes:['matters:read:summary','readiness:read:scoped'],timeoutSecs:600});
  const url=new URL(result.authorizationUrl);if(url.origin!=='https://app.rillora.xyz'||url.pathname!=='/api/ai/oauth/auth'||url.searchParams.get('code_challenge_method')!=='S256'||url.searchParams.get('client_id')!=='rillora-codex-v1'||url.searchParams.getAll('resource').length!==1||url.searchParams.get('resource')!==resource)throw Error('Unexpected client request');
  await writeFile(resolve(privateDir,'codex-authorization-url.txt'),url.href,{mode:0o600});
  const safe={source:'actual installed Codex',issuer:url.origin,clientId:url.searchParams.get('client_id'),resource,redirectUri:url.searchParams.get('redirect_uri'),pkce:'S256',scopes:url.searchParams.get('scope')};
  await writeFile(resolve(qa,'codex-authorization-request.json'),JSON.stringify(safe,null,2)+'\n');console.log(JSON.stringify(safe));
 }else if(cmd==='status'||cmd==='restart-status'){
  if(cmd==='restart-status'){child.kill();await new Promise(r=>child.once('exit',r));await start();}
  const result=await rpc('mcpServerStatus/list',{serverName:name,detail:'toolsAndAuthOnly'});
  await writeFile(resolve(qa,cmd==='status'?'codex-connected.json':'codex-after-revoke.json'),JSON.stringify(result,null,2)+'\n');console.log(JSON.stringify(result));
 }else if(cmd==='stop'){child.kill();break;}
}catch(error){console.log(JSON.stringify({failed:cmd,message:error.message}));}
child?.kill();
