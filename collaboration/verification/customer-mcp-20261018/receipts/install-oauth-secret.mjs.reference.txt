import {generateKeyPairSync,randomBytes,randomUUID} from 'node:crypto';
import {execFileSync} from 'node:child_process';
import {readFile,writeFile} from 'node:fs/promises';
const qa='output/qa/rillora-customer-mcp-20261018/',privateDir='.local-secrets/mcp-customer-20261018/';
if(!JSON.parse(await readFile(qa+'production-preparation-verified.json','utf8')).passed)throw Error('Preparation not verified');
function aws(args,stdin){return JSON.parse(execFileSync('aws',[...args,'--profile','rillora-production-operator','--region','eu-central-1','--output','json'],{input:stdin,encoding:'utf8',stdio:['pipe','pipe','pipe']}));}
const stack=aws(['cloudformation','describe-stacks','--stack-name','RilloraProduction']).Stacks[0];
if(stack.StackStatus!=='UPDATE_COMPLETE')throw Error('Stack not stable');
const arn=stack.Outputs.find(o=>o.OutputKey.startsWith('RuntimeCustomerMcpConfigurationSecretArn'))?.OutputValue;
if(!/^arn:aws:secretsmanager:eu-central-1:133857166372:secret:rillora\/production\/ai-oauth-[A-Za-z0-9]{6}$/.test(arn??''))throw Error('Wrong secret destination');
let intent;try{intent=JSON.parse(await readFile(privateDir+'oauth-secret-intent.json','utf8'));}catch(e){if(e.code!=='ENOENT')throw e;}
if(intent){
 if(intent.arn!==arn)throw Error('Secret destination changed');
 const state=aws(['secretsmanager','describe-secret','--secret-id',arn]);
 if(!state.VersionIdsToStages?.[intent.version]?.includes('AWSCURRENT'))throw Error('Uncertain prior key write requires inspection; refusing replacement');
}else{
 intent={arn,version:randomUUID(),createdAt:new Date().toISOString()};
 await writeFile(privateDir+'oauth-secret-intent.json',JSON.stringify(intent),{mode:0o600});
 const {privateKey}=generateKeyPairSync('rsa',{modulusLength:3072});
 const material={version:1,storageKey:randomBytes(32).toString('hex'),cookieKeys:[randomBytes(32).toString('hex')],
  jwks:{keys:[{...privateKey.export({format:'jwk'}),kid:randomUUID(),alg:'RS256',use:'sig'}]}};
 const input=Buffer.from(JSON.stringify(material));
 try{
  const result=aws(['secretsmanager','put-secret-value','--secret-id',arn,'--client-request-token',intent.version,'--secret-string','file:///dev/stdin'],input);
  if(result.VersionId!==intent.version||!result.VersionStages.includes('AWSCURRENT'))throw Error('Secret write not confirmed');
 }finally{input.fill(0);material.storageKey='';material.cookieKeys=[];material.jwks.keys=[];}
}
await writeFile(qa+'oauth-secret-installed.json',JSON.stringify({installed:true,awsManagedEncryption:true,rawKeysWrittenToDisk:false,secret:arn,version:intent.version,createdAt:intent.createdAt},null,2)+'\n');
console.log('Dedicated OAuth configuration protected in AWS; no key material printed or saved to source');
