import {readFile,writeFile} from 'node:fs/promises';import {execFileSync} from 'node:child_process';import {randomBytes} from 'node:crypto';
const qa='output/qa/rillora-customer-mcp-20261018/';
const args=['--profile','rillora-cdk-deploy','--region','eu-central-1','--output','json'];
const aws=a=>JSON.parse(execFileSync('aws',[...a,...args],{encoding:'utf8'}));
const stack=aws(['cloudformation','describe-stacks','--stack-name','RilloraProduction']).Stacks[0];if(stack.StackStatus!=='UPDATE_COMPLETE')throw Error('Update incomplete');
const raw=aws(['cloudformation','get-template','--stack-name','RilloraProduction']).TemplateBody;
const template=typeof raw==='string'?JSON.parse(raw):raw;
const expected=JSON.parse(await readFile(qa+'production-cdk-hostfix/RilloraProduction.template.json','utf8'));
function canonical(v){return JSON.stringify(v,(_,x)=>x&&typeof x==='object'&&!Array.isArray(x)?Object.fromEntries(Object.entries(x).sort(([a],[b])=>a.localeCompare(b))):x)}
if(canonical(template)!==canonical(expected))throw Error('Template mismatch');
const results=[];
for(const [name,path,options,status] of [
 ['health','/api/health/ready',{},200],['metadata','/api/mcp/resource-metadata',{},200],['issuer','/api/ai/oauth/.well-known/openid-configuration',{},200],
 ['anonymous','/api/mcp',{},401],['unknown-token','/api/mcp',{headers:{authorization:'Bearer '+randomBytes(32).toString('base64url')}},401],
 ['foreign-origin','/api/mcp',{method:'POST',headers:{origin:'https://untrusted.example.test','content-type':'application/json'},body:'{}'},403],
 ['userinfo-anonymous','/api/ai/oauth/userinfo',{},401]]){
 const r=await fetch('https://app.rillora.xyz'+path,{...options,redirect:'manual',signal:AbortSignal.timeout(30000)});const body=await r.json();
 if(r.status!==status)throw Error(name+' returned '+r.status);
 results.push({name,status:r.status,cache:r.headers.get('cache-control'),challenge:r.headers.get('www-authenticate'),body});
}
await writeFile(qa+'hostfix-verified.json',JSON.stringify({passed:true,stackStatus:stack.StackStatus,exactTemplate:true,at:new Date().toISOString(),results},null,2)+'\n');console.log('Exact routing fix passed: healthy, real DynamoDB rejection path, no anonymous access and strict origin protection');
