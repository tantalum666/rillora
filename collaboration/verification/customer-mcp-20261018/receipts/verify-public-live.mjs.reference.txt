import {readFile,writeFile} from 'node:fs/promises';import {createHash} from 'node:crypto';
const qa='output/qa/rillora-customer-mcp-20261018/';
const manifest=JSON.parse(await readFile('apps/web/dist-public/public-build.json','utf8'));
const route=Object.fromEntries(manifest.pages.map(p=>[p.file,p.path]));route['index.html']='/';
const results=[];const entries=Object.entries(manifest.files);let index=0;
async function worker(){while(index<entries.length){const [file,digest]=entries[index++];let last;
 for(let attempt=0;attempt<2;attempt++){try{
  const url='https://www.rillora.xyz'+(route[file]??'/'+file);const r=await fetch(url,{signal:AbortSignal.timeout(30000)});
  const bytes=Buffer.from(await r.arrayBuffer());const sha=createHash('sha256').update(bytes).digest('hex');
  if(r.status!==200||sha!==digest)throw Error(`Unexpected response for ${file}: ${r.status}`);
  results.push({file,url,status:r.status,sha256:sha});last=null;break;
 }catch(e){last=e;}}
 if(last)throw last;
}}
await Promise.all([worker(),worker(),worker(),worker()]);
await writeFile(qa+'public-live-http.json',JSON.stringify({passed:true,files:results.length,checkedAt:new Date().toISOString(),results:results.sort((a,b)=>a.file.localeCompare(b.file))},null,2)+'\n');
console.log('All '+results.length+' public files match their deployed canonical URLs');
