import {readFile,writeFile} from 'node:fs/promises';import {createHash} from 'node:crypto';
const qa='output/qa/rillora-customer-mcp-20261018/',base='https://app.rillora.xyz';const marker=JSON.parse(await readFile('apps/web/dist-production-live/production-build.json','utf8'));
const rows=[],items=Object.entries(marker.files);let next=0;
async function worker(){while(next<items.length){const [file,expected]=items[next++];const path=file==='index.html'?'/':('/'+file);let failure;
 for(let a=0;a<2;a++)try{const r=await fetch(base+path,{signal:AbortSignal.timeout(30000)});const data=Buffer.from(await r.arrayBuffer());const hash=createHash('sha256').update(data).digest('hex');if(r.status!==200||hash!==expected)throw Error('File mismatch '+file+' '+r.status);rows.push({file,status:r.status,sha256:hash});failure=null;break;}catch(e){failure=e;}
 if(failure)throw failure;
}}
await Promise.all([worker(),worker(),worker(),worker()]);
const discovery=[];
for(const path of ['/.well-known/oauth-authorization-server/api/ai/oauth','/.well-known/openid-configuration/api/ai/oauth','/.well-known/oauth-protected-resource/api/mcp']){
 const r=await fetch(base+path,{signal:AbortSignal.timeout(30000)});const body=await r.json();if(r.status!==200||r.headers.get('cache-control')!=='no-store')throw Error('Discovery failure');
 if(body.issuer&&body.issuer!==base+'/api/ai/oauth')throw Error('Wrong issuer');if(body.resource&&body.resource!==base+'/api/mcp')throw Error('Wrong resource');discovery.push({path,status:r.status,cache:r.headers.get('cache-control'),issuer:body.issuer,resource:body.resource});
}
await writeFile(qa+'web-live-http.json',JSON.stringify({passed:true,at:new Date().toISOString(),files:rows.length,downloadVerified:rows.some(r=>r.file==='integrations/rillora-ai-v1.zip'),discovery,rows:rows.sort((a,b)=>a.file.localeCompare(b.file))},null,2)+'\n');console.log('Verified '+rows.length+' public client files, download, and all three exact OAuth discovery routes');
