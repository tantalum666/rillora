from pathlib import Path
import subprocess,plistlib,json,hashlib,datetime
root=Path(__file__).resolve().parents[3]
q=root/'output/qa/rillora-device-config-20261006'
rel=root/'output/release/rillora-device-config-20261006'
ipa=rel/'export8/Rillora.ipa'
inspect=q/'inspection/build8'
inspect.mkdir(parents=True,exist_ok=True)
subprocess.run(['ditto','-x','-k',str(ipa),str(inspect)],check=True)
app=inspect/'Payload/Rillora.app'
info=plistlib.loads((app/'Info.plist').read_bytes())
baseline=plistlib.loads((rel/'Rillora-Review6.xcarchive/Products/Applications/Rillora.app/Info.plist').read_bytes())
verification=subprocess.run(['codesign','--verify','--deep','--strict',str(app)],capture_output=True,text=True)
ent=plistlib.loads(subprocess.check_output(['codesign','-d','--entitlements',':-',str(app)],stderr=subprocess.DEVNULL))
profile=plistlib.loads(subprocess.check_output(['security','cms','-D','-i',str(app/'embedded.mobileprovision')]))
checks={
 'strictSignature':verification.returncode==0,
 'correctVersion':info['CFBundleShortVersionString']=='1.0' and info['CFBundleVersion']=='8',
 'bundle':info['CFBundleIdentifier']=='com.rillora.ios',
 'team':ent.get('com.apple.developer.team-identifier')=='8L99CFNK24',
 'notDebuggable':ent.get('get-task-allow') is False,
 'noDeviceList':'ProvisionedDevices' not in profile,
 'notEnterprise':not profile.get('ProvisionsAllDevices',False),
 'noStoreKitFixture':not list(app.rglob('*.storekit')),
 'exactRetainedProductionConfiguration':all(info[k]==v for k,v in baseline.items() if k.startswith('Rillora') or k=='CFBundleURLTypes'),
 'bilingual':all((app/(language+'.lproj')).is_dir() for language in ['en','zh-Hans']),
 'approvedCryptoMetadata':info.get('ITSAppUsesNonExemptEncryption') is False,
 'privacyUnchanged':(app/'PrivacyInfo.xcprivacy').read_bytes()==(rel/'Rillora-Review6.xcarchive/Products/Applications/Rillora.app/PrivacyInfo.xcprivacy').read_bytes(),
 'sourceInputsMatch':all(hashlib.sha256((root/i['path']).read_bytes()).hexdigest()==i['sha256'] for i in json.loads((q/'native-build8-inputs.json').read_text())['inputs'])
}
result={'at':datetime.datetime.now(datetime.timezone.utc).isoformat(),'version':'1.0(8)','ipa':str(ipa.relative_to(root)),'bytes':ipa.stat().st_size,'sha256':hashlib.sha256(ipa.read_bytes()).hexdigest(),'checks':checks,'uploaded':False,'physicalFixVerified':False,'ownerExistingHkdPurchasePreserved':True}
(q/'distribution-build8.json').write_text(json.dumps(result,indent=2)+'\n')
print(json.dumps({'checks':checks,'passed':all(checks.values()),'ipaSha256':result['sha256']}))
assert all(checks.values())
