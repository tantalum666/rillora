for filename,data in [('final-operations-before-stack.json',stack),('final-operations-before-template.json',before)]:
 path=p/filename
 if path.exists():assert json.loads(path.read_text())==data
 else:path.write_text(json.dumps(data));path.chmod(0o600)
outputs={r['OutputKey']:r['OutputValue'] for r in stack['Outputs']}

 assert r.returncode==0
wanted=json.loads((assembly/(name+'.template.json')).read_text());old=before['Resources'];new=wanted['Resources']

added=set(new)-set(old)
assert len(added)==4 and all(k.startswith('AppleDelivery') and new[k]['Type']=='AWS::CloudWatch::Alarm' for k in added)
changes=[]
for key,value in new.items():

 if previous==value:continue
 if previous is None:pass
 elif value['Type']=='AWS::Lambda::Function':
  assert key.startswith('HealthCheck')
  expected=json.loads(json.dumps(previous));expected['Properties']['Code']=value['Properties']['Code']
  expected.get('Metadata',{})['aws:asset:path']=value.get('Metadata',{}).get('aws:asset:path')

  for target in old_config['targets']:
   if target['id'] in ['production-accounts-closed','production-client-accounts-closed']:target['expectedStatus']=401
  assert old_config==new_config
  expected['Properties']['Environment']['Variables']['OPERATIONS_TARGETS_JSON']=value['Properties']['Environment']['Variables']['OPERATIONS_TARGETS_JSON']
  assert expected==value
 elif value['Type'] in ['AWS::SNS::TopicPolicy','AWS::IAM::Policy']:
  # Only the four newly created alarm references may be added to read/publish scopes.
  def without_new_refs(v):
   if isinstance(v,list):return [without_new_refs(x) for x in v if not (isinstance(x,dict) and x.get('Fn::GetAtt',[None])[0] in added)]
   if isinstance(v,dict):return {k:without_new_refs(x) for k,x in v.items()}
   return v
  assert without_new_refs(value)==previous,'Unexpected permission change: '+key
 else:raise AssertionError('Unexpected operations change: '+key)
 changes.append({'logicalId':key,'type':value['Type'],'added':previous is None})
