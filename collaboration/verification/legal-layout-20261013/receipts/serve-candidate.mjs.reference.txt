import http from 'node:http';
import { readFile } from 'node:fs/promises';
const source = await readFile(process.argv[2] || '.local-secrets/legal-layout-20261013/pages/_worker.js');
const worker = (await import('data:text/javascript;base64,' + source.toString('base64'))).default;
http.createServer(async (request, response) => {
  try {
    const result = await worker.fetch(new Request('http://127.0.0.1:4281' + request.url, {
      method: request.method, headers: request.headers,
    }));
    response.writeHead(result.status, Object.fromEntries(result.headers));
    response.end(Buffer.from(await result.arrayBuffer()));
  } catch {
    response.writeHead(500);
    response.end('Candidate preview error');
  }
}).listen(4281, '127.0.0.1', () => console.log('Legal candidate ready on http://127.0.0.1:4281'));
