import assert from 'node:assert/strict';
import fs from 'node:fs/promises';
import { resolve } from 'node:path';
import { clientPackages } from '../../../tools/mcp/client-packages.mjs';
import { publicPages, publicRoute } from '../../../apps/web/src/publicSeo.js';

const root = process.cwd();
const read = async p => JSON.parse(await fs.readFile(resolve(root, p), 'utf8'));
const aso = await read('config/marketing/aso-vnext.json');
assert.equal(aso.status, 'VNEXT_DRAFT_NOT_SUBMITTED');
assert.equal(aso.currentReviewUnchanged, true);
const checked = [];
for (const locale of aso.locales) {
  for (const [field, limit] of Object.entries({name:30,subtitle:30,keywords:100,promotionalText:170,description:4000})) {
    assert.ok([...locale[field]].length > 0 && [...locale[field]].length <= limit, `${locale.locale} ${field}`);
  }
  assert.ok(Buffer.byteLength(locale.keywords, 'utf8') <= 100);
  assert.equal(new Set(locale.keywords.split(',')).size, locale.keywords.split(',').length);
  assert.doesNotMatch(locale.keywords, /Rillora|ChatGPT|Claude|Google|Apple/i);
  for (const field of ['supportUrl','marketingUrl']) {
    const url = new URL(locale[field]);
    assert.equal(url.origin, 'https://www.rillora.xyz');
    assert.ok(Object.hasOwn(publicPages, publicRoute(url.pathname).path));
  }
  checked.push({locale:locale.locale, ...locale.counts});
}
const publicIds = await read('integrations/ai/candidate-client-ids.json');
const generated = clientPackages(publicIds);
for (const [path, expected] of Object.entries(generated)) {
  assert.equal(await fs.readFile(resolve(root,'integrations/ai/hosted-candidate',path),'utf8'), expected, path);
}
const submission = await read('integrations/ai/submission-draft-20261018.json');
assert.equal(submission.positiveTests.length,5);
assert.equal(submission.negativeTests.length,3);
assert.equal(submission.productionEndpoint,null);
assert.equal(submission.directoryApproved,false);
for (const test of [...submission.positiveTests,...submission.negativeTests]) assert.equal(test.result,'NOT_RUN_IN_SUBMISSION_CLIENT');
const sample = clientPackages(await read('integrations/ai/example-client-ids.json'));
assert.equal(JSON.parse(sample['readiness.json']).status,'EXAMPLE_ONLY_NOT_INSTALLABLE');
const result = {at:new Date().toISOString(),passed:true,aso:checked,hostedPackageFilesMatched:Object.keys(generated).length,
  currentlyRegisteredClients:Object.keys(publicIds.clientIds),examplesRemainNotInstallable:true,submissionDraftCases:8,
  naturalLanguageModelCalls:0,installedClientsChanged:false,deployed:false};
await fs.writeFile(resolve(root,'output/qa/rillora-marketing-readiness-20261018/materials-check.json'),JSON.stringify(result,null,2)+'\n');
console.log(JSON.stringify(result));
