"""Build only the two changed Web clients; retain approved legal inputs privately."""
from pathlib import Path
import datetime,json,os,re,shlex,subprocess
ROOT=Path(__file__).resolve().parents[3]
QA=ROOT/'output/qa/rillora-native-cross-platform-20261012'
cfg=Path('/Users/tantalum/Documents/电子遗嘱app/.local-secrets/legal/release.env')
assert cfg.stat().st_uid==os.getuid() and cfg.stat().st_mode&0o777==0o600
values={}
for raw in cfg.read_text().splitlines():
    line=raw.strip().removeprefix('export ').strip()
    if not line or line.startswith('#'):continue
    key,value=line.split('=',1)
    assert re.fullmatch(r'VITE_LEGAL_[A-Z_]+',key)
    assert '$(' not in value and '`' not in value and '${' not in value
    parts=shlex.split(value);assert len(parts)==1;values[key]=parts[0]
assert len(values)==6
env={**os.environ,**values,'VITE_API_BASE_URL':'/api/v1','RILLORA_CLOSED_CLIENT_CANDIDATE':'1'}
results=[]
for mode in ['build:release','build:production-review']:
    path=QA/(mode.replace(':','-')+'-final.log')
    assert not path.exists(),'Inspect previous build rather than overwrite it'
    with path.open('w') as log:
        result=subprocess.run(['npm','run',mode],cwd=ROOT/'apps/web',env=env,stdout=log,stderr=subprocess.STDOUT)
    results.append({'build':mode,'exitCode':result.returncode})
    assert result.returncode==0,'Inspect the scoped build log'
(QA/'web-final-builds.json').write_text(json.dumps({'at':datetime.datetime.now(datetime.timezone.utc).isoformat(),'results':results,'apiAndNativeRebuilt':False,'legalInputsChanged':False},indent=2)+'\n')
print(json.dumps(results))
