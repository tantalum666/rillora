import http from 'node:http';
import {readFile} from 'node:fs/promises';
const source=await readFile('.local-secrets/payment-recheck-20261009/legal-pages-package/_worker.js');
const worker=(await import('data:text/javascript;base64,'+source.toString('base64'))).default;
const server=http.createServer(async(req,res)=>{try{const response=await worker.fetch(new Request('http://127.0.0.1:4276'+req.url,{method:req.method,headers:req.headers}));res.writeHead(response.status,Object.fromEntries(response.headers));res.end(Buffer.from(await response.arrayBuffer()));}catch{res.writeHead(500);res.end('Local preview error');}});
server.listen(4276,'127.0.0.1',()=>console.log('Reviewed legal candidate on http://127.0.0.1:4276'));
