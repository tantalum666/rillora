import {readFile,writeFile,stat} from 'node:fs/promises';
import {X509Certificate,createHash} from 'node:crypto';
import {AppStoreServerAPIClient,Environment} from '../../../apps/api/node_modules/@apple/app-store-server-library/dist/index.js';
const root=new URL('../../../',import.meta.url);
async function secret(name){const p=new URL('.local-secrets/storekit/'+name,root);if((await stat(p)).mode&0o077)throw new Error('Credential permissions');return readFile(p,'utf8');}
const client=new AppStoreServerAPIClient(await secret('iap-private-key.p8'),(await secret('key-id')).trim(),(await secret('issuer-id')).trim(),'com.rillora.ios',Environment.SANDBOX);
const receipt=JSON.parse(await readFile(new URL('apple-test-notification.json',import.meta.url),'utf8'));
const result=await client.getTestNotificationStatus(receipt.testNotificationToken);
const sha=createHash('sha256').update(result.signedPayload).digest('hex');
if(!receipt.observations.some(x=>x.signedPayloadSha256===sha))throw new Error('Unexpected TEST payload change');
const header=JSON.parse(Buffer.from(result.signedPayload.split('.')[0],'base64url'));
const endpoints=header.x5c.slice(0,2).map(value=>new X509Certificate(Buffer.from(value,'base64')).infoAccess?.match(/^OCSP - URI:(.*)$/m)?.[1]);
if(endpoints.some(x=>!x))throw new Error('Missing OCSP endpoint');
const dns=[];
for(const host of new Set(endpoints.map(x=>new URL(x).hostname))){
 const response=await fetch('https://dns.google/resolve?name='+encodeURIComponent(host)+'&type=A');if(!response.ok)throw new Error('DNS lookup failed');
 const body=await response.json();const addresses=(body.Answer??[]).filter(x=>x.type===1).map(x=>x.data);
 dns.push({host,addresses,allWithinApple17Block:addresses.length>0&&addresses.every(x=>x.startsWith('17.'))});
}
const record={at:new Date().toISOString(),source:'Exact Apple TEST notification that passed local online verification',signedPayloadSha256:sha,endpoints,dns,cloudFailure:{verificationStatus:2,cause:'ETIMEDOUT'},existingCloudEgress:'TCP443 only, plus explicit database traffic',proposedChange:'Permit TCP80 only to Apple-owned 17.0.0.0/8 for OCSP; keep online certificate and signed response checks enabled'};
await writeFile(new URL('apple-ocsp-transport-evidence.json',import.meta.url),JSON.stringify(record,null,2)+'\n');console.log(JSON.stringify(record));
