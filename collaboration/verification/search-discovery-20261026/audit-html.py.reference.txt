from pathlib import Path
from html.parser import HTMLParser
from urllib.parse import urljoin,urlsplit
from collections import Counter,deque
import json,re,sys
qa=Path(__file__).resolve().parent
class Page(HTMLParser):
 def __init__(self):
  super().__init__(); self.stack=[]; self.text=[]; self.links=[]; self.headings=[]; self.meta=[]; self.canonical=[]; self.alternates=[]; self.ids=[]; self.title=[]; self.lang=None; self.assets=[]
 def handle_starttag(self,tag,attrs):
  a=dict(attrs)
  if tag=='html':self.lang=a.get('lang')
  if a.get('id'):self.ids.append(a['id'])
  if tag=='meta':self.meta.append(a)
  if tag=='link' and a.get('rel')=='canonical':self.canonical.append(a.get('href'))
  if tag=='link' and a.get('rel')=='alternate':self.alternates.append(a)
  if tag=='a':self.links.append({'href':a.get('href',''),'inMain':'main' in self.stack,'text':''})
  if tag in ['h1','h2','h3']:self.headings.append({'level':tag,'text':''})
  if tag in ['script','link','img','source']:
   for k in ['src','href','srcset']:
    if a.get(k,'').startswith('/assets/'):self.assets.append(a[k])
  if tag not in ['meta','link','img','source','input','br','hr']:self.stack.append(tag)
 def handle_endtag(self,tag):
  if tag in self.stack:
   pos=len(self.stack)-1-self.stack[::-1].index(tag);self.stack=self.stack[:pos]
 def handle_data(self,d):
  if 'title' in self.stack:self.title.append(d)
  if 'main' in self.stack and not {'script','style','svg'}&set(self.stack):self.text.append(d)
  if self.links and 'a' in self.stack:self.links[-1]['text']+=d
  if self.headings and any(h in self.stack for h in ['h1','h2','h3']):self.headings[-1]['text']+=d

 p=Page(); h=(qa/r['file']).read_text(); p.feed(h); text=' '.join(' '.join(p.text).split())

known={r['url'] for r in rows}; edges={r['url']:{urljoin(r['url'],a['href']).split('#')[0].split('?')[0] for a in r['links'] if a['href']} for r in rows}
for r in rows:
 r['inboundPages']=sorted(u for u,v in edges.items() if r['url'] in v and u!=r['url'])

 broken=[]
