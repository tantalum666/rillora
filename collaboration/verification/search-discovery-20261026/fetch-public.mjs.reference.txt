import https from 'node:https';
import {readFile,writeFile,mkdir} from 'node:fs/promises';
const qa=new URL('./',import.meta.url);await mkdir(new URL('before-html/',qa),{recursive:true});
const source=JSON.parse(await readFile(new URL('google-27.json',qa),'utf8'));
const urls=['https://www.rillora.xyz/en/',...source.urls];
const get=url=>new Promise((resolve,reject)=>{const start=Date.now();const r=https.get(url,{agent:false,rejectUnauthorized:true,headers:{'user-agent':'Rillora-SEO-Audit/1.0','accept':'text/html'}},res=>{let body='';res.setEncoding('utf8');res.on('data',d=>body+=d);res.on('end',()=>resolve({url,status:res.statusCode,elapsedMs:Date.now()-start,headers:Object.fromEntries(['content-type','content-security-policy','x-robots-tag','cache-control','location'].filter(k=>res.headers[k]).map(k=>[k,res.headers[k]])),body}));});r.setTimeout(20000,()=>r.destroy(new Error('timeout')));r.on('error',reject);});
const rows=[];for(let i=0;i<urls.length;i+=4){const r=await Promise.allSettled(urls.slice(i,i+4).map(get));for(let j=0;j<r.length;j++){if(r[j].status==='fulfilled'){const {body,...row}=r[j].value;row.file=`before-html/${String(i+j).padStart(2,'0')}.html`;await writeFile(new URL(row.file,qa),body);rows.push(row);}else rows.push({url:urls[i+j],error:r[j].reason.message});}}
await writeFile(new URL('before-http.json',qa),JSON.stringify({checkedAt:new Date().toISOString(),boundary:'Current anonymous normal HTTP GET, not Googlebot',rows},null,2)+'\n');console.log(JSON.stringify({pages:rows.length,statuses:rows.reduce((m,r)=>(m[r.status||r.error]=(m[r.status||r.error]||0)+1,m),{})}));
