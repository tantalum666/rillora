import https from 'node:https';import {readFile,writeFile} from 'node:fs/promises';import {createHash} from 'node:crypto';
const qa=new URL('./',import.meta.url);const phase=process.argv[2];const root=new URL('../../../',qa);const candidate=JSON.parse(await readFile(new URL('apps/web/dist-public/public-build.json',root)));const old=JSON.parse(await readFile(new URL('old-public-build.json',qa)));const hash=v=>createHash('sha256').update(v).digest('hex');
const get=path=>new Promise((resolve,reject)=>{const req=https.get('https://www.rillora.xyz'+path,{agent:false,rejectUnauthorized:true,headers:{'user-agent':'Rillora-Release-Readback/1.0'}},res=>{const chunks=[];res.on('data',d=>chunks.push(d));res.on('end',()=>resolve({status:res.statusCode,body:Buffer.concat(chunks),headers:res.headers}));});req.on('error',reject);req.setTimeout(25000,()=>req.destroy(new Error('timeout')));});
const expected=phase==='assets'?old:candidate;const paths=new Map(expected.pages.map(p=>[p.file,p.path]));paths.set('index.html','/');const checks=[];
if(phase==='assets'){
 for(const [file,digest] of Object.entries(old.files))if(!file.startsWith('assets/'))checks.push({file,path:paths.get(file)||'/'+file,digest,group:'old-pages-and-discovery'});
 for(const [file,digest] of Object.entries(candidate.files))if(file.startsWith('assets/'))checks.push({file,path:'/'+file,digest,group:'candidate-assets'});
}else for(const [file,digest] of Object.entries(candidate.files))checks.push({file,path:paths.get(file)||'/'+file,digest,group:'candidate'});
const results=[];for(let i=0;i<checks.length;i+=4){const batch=await Promise.allSettled(checks.slice(i,i+4).map(async c=>{const r=await get(c.path);return {...c,status:r.status,hashMatches:hash(r.body)===c.digest,csp:r.headers['content-security-policy']||null,hasCookie:!!r.headers['set-cookie']};}));for(let j=0;j<batch.length;j++)results.push(batch[j].status==='fulfilled'?batch[j].value:{...checks[i+j],error:batch[j].reason.message});}
const failures=results.filter(r=>r.status!==200||!r.hashMatches||r.hasCookie||!r.csp?.includes("connect-src 'none'"));const output={at:new Date().toISOString(),summary:{count:results.length,passed:failures.length===0,failures},results};await writeFile(new URL(phase+'-http.json',qa),JSON.stringify(output,null,2)+'\n');console.log(JSON.stringify(output.summary));if(failures.length)process.exitCode=1;
