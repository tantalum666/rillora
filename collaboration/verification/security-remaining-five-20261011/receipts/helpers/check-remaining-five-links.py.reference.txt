PRIMARY=Path('/Users/tantalum/Documents/电子遗嘱app')

 links=re.findall(r'\]\((<[^>]+>|[^\n)]+)\)',body)+re.findall(r'^\s*\[[^]]+\]:\s*(\S+)',body,re.M)
 for raw in links:

  if not target or target.startswith('#') or re.match(r'^[a-zA-Z][a-zA-Z+.-]*:',target):continue
