if '--ack' in sys.argv and pending.exists():
 for item in json.loads(pending.read_text()):
  row=next(r for r in data['files'] if r['path']==item['path']);assert hashlib.sha256((root/row['path']).read_bytes()).hexdigest()==row['sha256'];row['readRanges'].append([item['start'],item['end']])

 print(json.dumps({'readFiles':sum(bool(r['readRanges']) and max(t[1] for t in r['readRanges'])==r['lines'] for r in data['files']),'totalFiles':len(data['files']),'readLines':sum(b-a+1 for r in data['files'] for a,b in r['readRanges']),'totalLines':sum(r['lines'] for r in data['files'])}));raise SystemExit
budget=int(sys.argv[sys.argv.index('--chars')+1]) if '--chars' in sys.argv else 10000
cost=(lambda s: len(s)) if '--chars' in sys.argv else (lambda s: len(s)/3+len(re.findall(r'[\u3400-\u9fff]',s))*0.9)

 for index in range(start-1,len(lines)):

  if used+sum(map(cost,chosen))+cost(line)>budget and (chosen or emitted):break

 if not chosen:break
 body=''.join(chosen);used+=cost(body);print(f'\nFILE {row["path"]} L{start}-{end}/{row["lines"]}\n{body}',end='');emitted.append({'path':row['path'],'start':start,'end':end})
 if end<row['lines'] or used>=budget:break

 data['completedAt']=datetime.datetime.now(datetime.timezone.utc).isoformat();ledger.write_text(json.dumps(data,ensure_ascii=False,indent=2)+'\n');print('FULL MARKDOWN READING COMPLETE')
