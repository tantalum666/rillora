async (page) => {
  const checks = [];
  const errors = [];
  const recordError = error => errors.push(error.message);
  page.on('pageerror', recordError);
  const check = (value, label) => { if (!value) throw new Error(label); checks.push(label); };
  const button = name => page.getByRole('button', { name, exact: true });
  const root = 'http://127.0.0.1:4413/';
  try {
    page.setDefaultTimeout(5000);
    await page.setViewportSize({ width: 1440, height: 1000 });
    await page.goto(root);
    await page.getByRole('heading', { name: 'Your map', exact: true }).waitFor();
    check(await page.title() === 'Rillora — Your private map for what comes next', 'page identity');
    await button('Shared reviews').click();
    await page.getByRole('button', { name: /Review schedule & collaborator/ }).click();
    const cadence = page.getByRole('combobox', { name: 'Review cadence' });
    await cadence.selectOption('30');
    await page.goBack();
    await page.getByRole('heading', { name: 'Discard unsaved changes?' }).waitFor();
    await button('Keep editing').click();
    await page.waitForURL(url => url.searchParams.get('view') === 'maintenance');
    check(await cadence.inputValue() === '30', 'review draft survives Back and Keep editing');
    await page.goBack();
    await page.getByRole('heading', { name: 'Discard unsaved changes?' }).waitFor();
    await page.goForward();
    await page.getByRole('heading', { name: 'Discard unsaved changes?' }).waitFor({ state: 'hidden' });
    check(await cadence.inputValue() === '30', 'returning to original entry cancels stale discard decision');
    await page.screenshot({ path: '/Users/tantalum/Documents/电子遗嘱app/output/qa/rillora-strict-review-20261002/review-draft-after-desktop.png' });
    await page.setViewportSize({ width: 390, height: 844 });
    await page.goBack();
    await page.getByRole('heading', { name: 'Discard unsaved changes?' }).waitFor();
    await button('Keep editing').click();
    await page.waitForURL(url => url.searchParams.get('view') === 'maintenance');
    check(await cadence.inputValue() === '30', 'review draft survives mobile Back and Keep editing');
    await page.screenshot({ path: '/Users/tantalum/Documents/电子遗嘱app/output/qa/rillora-strict-review-20261002/review-draft-after-mobile.png' });
    await page.setViewportSize({ width: 1440, height: 1000 });
    await page.goBack();
    await button('Discard changes').click();
    await page.getByRole('heading', { name: 'Your map', exact: true }).waitFor();
    check(new URL(page.url()).search === '', 'discard returns to matching overview route');
    await button('Shared reviews').click();
    await page.getByRole('button', { name: /Review schedule & collaborator/ }).click();
    check(await cadence.inputValue() === '90', 'discard did not silently save review cadence');
    await button('Close review panel').click();
    await page.goBack();
    await page.getByRole('heading', { name: 'Your map', exact: true }).waitFor();
    await button('My plan').click();
    await button('Matters').click();
    await button('Add a matter').click();
    const name = page.getByRole('textbox', { name: 'Name', exact: true });
    await name.fill('Synthetic back navigation draft');
    const original = page.url();
    await page.goBack();
    await page.getByRole('heading', { name: 'Discard unsaved changes?' }).waitFor();
    await page.goBack();
    await button('Keep editing').click();
    await page.waitForURL(original);
    check(await name.inputValue() === 'Synthetic back navigation draft', 'repeated Back preserves the draft');
    check(new URL(page.url()).searchParams.get('task') === 'matter-create', 'repeated Back cancel restores exact form URL');
    await page.screenshot({ path: '/Users/tantalum/Documents/电子遗嘱app/output/qa/rillora-strict-review-20261002/repeated-back-after.png' });
    await page.goBack();
    await page.getByRole('heading', { name: 'Discard unsaved changes?' }).waitFor();
    await page.goBack();
    await button('Discard changes').click();
    await page.waitForURL(url => url.searchParams.get('view') === 'plan');
    await page.getByRole('heading', { name: 'My plan', exact: true }).waitFor();
    check(await page.getByRole('dialog').count() === 0, 'repeated Back discard reaches latest destination');
    await page.goForward();
    await page.getByRole('heading', { name: 'Your matters', exact: true }).waitFor();
    await page.goForward();
    await page.getByRole('heading', { name: 'New matter', exact: true }).waitFor();
    check(await name.inputValue() === '', 'Forward does not revive discarded private draft');
    await page.goBack();
    await page.getByRole('heading', { name: 'Your matters', exact: true }).waitFor();
    check(await page.getByRole('dialog').count() === 0, 'clean form Back needs no discard prompt');
    check(await page.locator('vite-error-overlay').count() === 0, 'no framework overlay');
    check(errors.length === 0, 'no runtime page errors');
    return { checks, errors, url: page.url(), viewports: ['1440x1000', '390x844'], syntheticOnly: true,
      browserAvailability: 'Browser plugin not available; Playwright CLI used' };
  } catch (error) {
    throw new Error(JSON.stringify({ completed: checks, url: page.url(), error: error.message }));
  } finally { page.off('pageerror', recordError); }
}
