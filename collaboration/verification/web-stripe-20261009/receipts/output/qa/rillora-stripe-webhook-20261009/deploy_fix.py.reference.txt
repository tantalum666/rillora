changes=[]
def diff(a,b,path=''):
    if type(a)!=type(b):changes.append(path)
    elif isinstance(a,dict):
        for k in sorted(a.keys()|b.keys()):
            if k not in a or k not in b:changes.append(path+'/'+k)
            else:diff(a[k],b[k],path+'/'+k)
    elif isinstance(a,list):
        if len(a)!=len(b):changes.append(path)
        else:
            for i,(x,y) in enumerate(zip(a,b)):diff(x,y,path+'/'+str(i))
    elif a!=b:changes.append(path)
diff(old,new)
assert sorted(changes)==sorted('/Resources/'+k+'/Properties/ContainerDefinitions/0/Image/Fn::Sub' for k in tasks)
