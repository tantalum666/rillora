def aws(args,profile='rillora-staging-operator'):
    result=subprocess.run(['aws',*args,'--profile',profile,'--region','eu-central-1','--output','json'],capture_output=True,text=True)
    if result.returncode:raise RuntimeError(result.stderr[:800])
    return json.loads(result.stdout)

script="""import {spawnSync} from 'node:child_process';
import {readFile,readdir} from 'node:fs/promises';
import {createHash} from 'node:crypto';
import pg from 'pg';
import {loadDatabaseConfig} from './dist/config.js';
import {databaseConnectionOptions} from './dist/db.js';
const cfg=loadDatabaseConfig();if(cfg.sslMode!=='verify-full')throw new Error('verified_tls_required');
if(process.env.ROTATE_RUNTIME_DATABASE_PASSWORD==='true')throw new Error('credential_rotation_not_requested');
for(const file of ['dist/migrate.js','dist/bootstrap-runtime.js']){
 const child=spawnSync(process.execPath,[file],{stdio:'inherit'});
 if(child.error||child.status!==0)process.exit(child.status||2);
}
const client=new pg.Client({...databaseConnectionOptions(cfg),application_name:'rillora-staging-migration-verification'});
await client.connect();
try {
 await client.query('begin read only');
 const tls=(await client.query('select ssl from pg_stat_ssl where pid=pg_backend_pid()')).rows[0]?.ssl;
 const rows=(await client.query('select name,checksum from app.schema_migrations order by name')).rows;
 const files=(await readdir('migrations')).filter(n=>/^\\d+_[a-z0-9_]+\\.sql$/.test(n)).sort();
 if(!tls||rows.length!==93||files.length!==93)throw new Error('unexpected_schema_or_tls');
 const byName=new Map(rows.map(r=>[r.name,r.checksum]));
 for(const name of files){if(byName.get(name)!==createHash('sha256').update(await readFile('migrations/'+name)).digest('hex'))throw new Error('migration_checksum_mismatch');}
 await client.query('commit');
 console.log(JSON.stringify({migrationSucceeded:true,schemaCount:rows.length,checksumsMatch:true,tls:true,runtimeCredentialRotation:false}));
}finally{await client.end();}
"""
