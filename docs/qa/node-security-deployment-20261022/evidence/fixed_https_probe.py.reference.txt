"""Bounded unauthenticated HTTPS metadata only, for the reviewed fixed deployment domains."""
import concurrent.futures
import datetime
import json
from pathlib import Path
import urllib.request
import urllib.error

class NoRedirect(urllib.request.HTTPRedirectHandler):
    def redirect_request(self,*args,**kwargs):return None
urls=[
 ('staging-live','https://l1g8350qw0.execute-api.eu-central-1.amazonaws.com/health/live',200),
 ('staging-ready','https://l1g8350qw0.execute-api.eu-central-1.amazonaws.com/health/ready',200),
 ('staging-session','https://app.staging.rillora.xyz/api/v1/session',401),
 ('production-live','https://j749rhzub7.execute-api.eu-central-1.amazonaws.com/health/live',200),
 ('production-ready','https://j749rhzub7.execute-api.eu-central-1.amazonaws.com/health/ready',200),
 ('production-session','https://app.rillora.xyz/api/v1/session',401),
 ('production-mcp-discovery','https://app.rillora.xyz/.well-known/oauth-protected-resource/api/mcp',200),
]
def probe(item):
    name,url,expected=item
    opener=urllib.request.build_opener(NoRedirect)
    request=urllib.request.Request(url,headers={'User-Agent':'Rillora-Runtime-Upgrade-Readback/1.0'})
    try:
        try:response=opener.open(request,timeout=15)
        except urllib.error.HTTPError as error:response=error
        with response:
            headers={k:response.headers.get(k) for k in ['cache-control','strict-transport-security','content-security-policy','x-content-type-options']}
            status=response.status
        return {'id':name,'url':url,'httpStatus':status,'expectedStatus':expected,'headers':headers,'passed':status==expected}
    except Exception as error:return {'id':name,'url':url,'httpStatus':None,'passed':False,'errorType':type(error).__name__}
with concurrent.futures.ThreadPoolExecutor(max_workers=7) as executor:checks=list(executor.map(probe,urls))
print(json.dumps({'checkedAt':datetime.datetime.now(datetime.timezone.utc).isoformat(),'scope':'fixed-public-https-metadata-only','cookiesOrCredentialsUsed':False,'responseBodiesStored':False,'tlsVerificationDisabled':False,'checks':checks,'passed':all(c['passed'] for c in checks)},indent=2))
